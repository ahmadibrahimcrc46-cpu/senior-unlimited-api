function openPage(id,btn){ document.querySelectorAll('.page').forEach(p=>p.classList.remove('active')); document.getElementById(id).classList.add('active'); document.querySelectorAll('.tabs button').forEach(b=>b.classList.remove('active')); btn.classList.add('active'); // Save to history so Back button works history.pushState({page:id}, "", "#"+id); localStorage.setItem('lastPage', id); } // Back button fix window.addEventListener('popstate', function(e){ let page = location.hash.replace('#','') || localStorage.getItem('lastPage') || 'home'; if(!page) page='home'; let btn = document.querySelector(`.tabs button[onclick*="'${page}'"]`); if(btn){ document.querySelectorAll('.page').forEach(p=>p.classList.remove('active')); document.getElementById(page)?.classList.add('active'); document.querySelectorAll('.tabs button').forEach(b=>b.classList.remove('active')); btn.classList.add('active'); } }); // On load, open last page window.addEventListener('load', ()=>{ let hash = location.hash.replace('#',''); if(hash){ let btn = document.querySelector(`.tabs button[onclick*="'${hash}'"]`); if(btn) openPage(hash, btn); } });
